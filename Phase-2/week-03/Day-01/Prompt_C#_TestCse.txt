Expected Selenium C# POM:

using OpenQA.Selenium;
using System.Collections.Generic;
using SonicFramework.Helpers;
using SonicFramework.Pages;

namespace SonicFramework.Pages
{
    public class LoginPage : SeleniumWrapper
    {
        private readonly Dictionary<string, string> Selectors;

        public LoginPage(IWebDriver driver) : base(driver)
        {
            Selectors = new Dictionary<string, string>
            {
                ["UsernameField"] = "//input[@id='username']",
                ["PasswordField"] = "password",
                ["LoginButton"] = "Login"
            };
        }

        public LoginPage EnterUname(string uName)
        {
            ClearAndType(
                Selectors["UsernameField"],
                "Username Field",
                uName);

            ReportStep("uName entered successfully", "pass");

            return this;
        }

        public LoginPage EnterPwd(string pwd)
        {
            ClearAndType(
                Selectors["PasswordField"],
                "Password Field",
                pwd);

            ReportStep("pwd entered successfully", "pass");

            return this;
        }

        public WelcomePage ClickLogin()
        {
            Click(
                Selectors["LoginButton"],
                "Login Button",
                "Button");

            ReportStep("logged in successfully", "pass");

            return new WelcomePage(driver);
        }
    }
}
